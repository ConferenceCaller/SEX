# Mary is appointed regent
country_event = {
	id = incidents_bur_inheritance.0
	title = "incidents_bur_inheritance.0.t"
	desc = "incidents_bur_inheritance.0.d"
	picture = DEATH_OF_HEIR_eventPicture
	
	fire_only_once = yes
	
	trigger = {
		tag = BUR
		has_country_modifier = burgundian_succession_crisis
		NOT = { has_country_flag = burgundian_succession_start }
		is_free_or_tributary_trigger = yes
		government = monarchy
		has_regency = no
		NOT = { is_year = 1500 }
		OR = {
			AND = {
				has_heir = no
				is_year = 1457
			}
			AND = {
				has_ruler = "Charles"
				dynasty = "de Bourgogne"
				ruler_age = 43
				NOT = {
					heir_age = 15
				}
			}
		}
	}
	
	mean_time_to_happen = {
		months = 120

		modifier = {
			factor = 0.5
			is_year = 1473
		}
	}
	
	option = {
		name = "incidents_bur_inheritance.0.a"
		ai_chance = { 
			factor = 100
		}
		if = {
			limit = {
				is_year = 1473
			}
			define_heir = {
				name = "Marie"
				dynasty = ROOT
				birth_date = 1457.02.13
				adm = 4
				dip = 5
				mil = 3
				claim = 95
				block_disinherit = yes
				female = yes
			}
		}
		else = {
			define_heir = {
				name = "Marie"
				dynasty = ROOT
				age = 16
				adm = 4
				dip = 5
				mil = 3
				claim = 95
				block_disinherit = yes
				female = yes
			}
		}

		set_heir_flag = mary_is_heir
	}
	option = {
		name = "incidents_bur_inheritance.0.b"
		ai_chance = { 
			factor = 0
		}
		add_legitimacy = 20
	}
}

# Ruler of Burgundy Dies
country_event = {
	id = incidents_bur_inheritance.1
	title = "incidents_bur_inheritance.1.t"
	desc = "incidents_bur_inheritance.1.desc"
	picture = DEATH_OF_HEIR_eventPicture
	
	major = yes
	
	is_triggered_only = yes
	fire_only_once = yes
	
	immediate = {
		hidden_effect = {
			set_country_flag = burgundian_succession_start
			add_country_modifier = { 
				name = mary_is_on_the_throne
				duration = 14600 
				hidden = yes
			}
		}
	}
	
	option = { #stand alone
		name = "incidents_bur_inheritance.1.a"
		ai_chance = {
			factor = 100
		}
		trigger = {
			FRA = { 
				exists = yes
				government = monarchy
				is_emperor = no
				is_subject = no
			}
			NOT = {
				any_known_country = {
					overlord_of = BUR
					OR = {
						is_in_war = {
							attacker_leader = FRA
							defender_leader = THIS
						}
						is_in_war = {
							defender_leader = FRA
							attacker_leader = THIS
						}
					}
				}
			}
		}
		custom_tooltip = bur_france_may_get_mad_tt
		custom_tooltip = lowland_privilege_tt
		hidden_effect = {
			FRA = {
				if = {
					limit = {
						alliance_with = BUR
					}
					break_alliance = BUR
					break_marriage = BUR
					add_casus_belli = {
						target = BUR
						type = cb_restore_personal_union
						months = 120
					}
					every_province = {
						limit = {
							owned_by = BUR
							region = france_region
						}
						add_core = FRA
					}
				}
				else = {
					country_event = {
						id = incidents_bur_inheritance.101
						days = 10
					}
				}
			}
			emperor = {
				if = {
					limit = {
						has_dlc = "Emperor"
					}
					set_imperial_incident = incident_burgundian_succ_alone
					emperor = {
						break_alliance = BUR
						break_marriage = BUR
					}
				}
				else = {
					country_event = {
						id = incidents_bur_inheritance.4
						days = 5
					}
				}
			}
		}
	}
	option = { #Stand alone, but France doesn't exist, France is subject itself or France IS the Emperor
		name = "incidents_bur_inheritance.1.b"
		ai_chance = {
			factor = 100
		}
		trigger = { 
			FRA = { 
				OR = {
					exists = no
					is_emperor = yes
					is_subject = yes
				}
			}
		}
		custom_tooltip = lowland_privilege_tt
		hidden_effect = {
			emperor = {
				if = {
					limit = {
						has_dlc = "Emperor"
					}
					set_imperial_incident = incident_burgundian_succ_alone
				}
				else = {
					country_event = {
						id = incidents_bur_inheritance.4
						days = 5
					}
				}
			}
		}
	}
}

# Emperor demands the Lowlands
country_event = {
	id = incidents_bur_inheritance.102
	title = "incidents_bur_inheritance.102.t"
	desc = "incidents_bur_inheritance.102.d"
	picture = DEATH_OF_HEIR_eventPicture

	is_triggered_only = yes
	fire_only_once = yes
	
	option = { #Let them come
		name = "incidents_bur_inheritance.102.a"
		ai_chance = {
			factor = 0
		}
		every_country = {
			limit = {
				is_subject_of = BUR
				OR = {
					primary_culture = dutch
					primary_culture = flemish
					primary_culture = frisian
					is_part_of_hre = yes
				}
			}
			emperor = { support_independence_of = PREV }
			if = {
				limit = {
					NOT = { tag = HOL }
				}
				create_alliance = HOL
			}
			if = {
				limit = {
					NOT = { is_part_of_hre = yes }
				}
				set_in_empire = yes
			}
		}
		hidden_effect = {
			BUR = {
				clr_country_flag = burgundian_succession_start
				set_country_flag = bur_crisis_has_happened
				clr_country_flag = burgundy_during_imperial_incident
			}
		}
	}
	option = { #Let the people go
		name = "incidents_bur_inheritance.102.c"
		ai_chance = { 
			factor = 100
		}
		custom_tooltip = incidents_bur_inheritance.102.c.tt
		hidden_effect = {
			every_country = {
				limit = {
					capital_scope = {
						OR = {
							culture = dutch
							culture = flemish
							culture = frisian
						}
						region = low_countries_region
						is_part_of_hre = yes
					}
					is_subject_of = BUR
				}
				grant_independence = yes #vassals gain independence
				if = {
					limit = {
						is_part_of_hre = no
					}
					set_in_empire = yes	
				}
			}
			BUR = {
				clr_country_flag = burgundian_succession_start
				set_country_flag = bur_crisis_has_happened
				clr_country_flag = burgundy_during_imperial_incident
			}
		}
	}
}
