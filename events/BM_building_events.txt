# General building deletion

country_event = {
	id = building.1
    title = building.1.t
    desc = building.1.d
	is_triggered_only = yes
	picture = CITY_DEVELOPMENT_eventPicture

	option = {
		name = building.1.remove_all_unnecessary
		trigger = { ROOT = { OR = { temple = 1 cathedral = 1 coastal_defence = 1 naval_battery = 1 dock = 1 drydock = 1 shipyard = 1 grand_shipyard = 1 } } }
        custom_tooltip = building_remove_all_unnecessary_TT
	}

	option = {
		name = building.1.manufactories
        ROOT = {
            country_event = { 
                id = building.1.1
                days = 0
            } 
        }
	}

	# Temple chain
	option = {
		name = building_tax_all
		trigger = { ROOT = { OR = { temple = 1 cathedral = 1 } } }
		tooltip = { custom_tooltip = building_tax_all_TT }
		delete_building_and_call_event_effect = { BUILDING = temple BUILDING = cathedral }
	}
	option = {
		name = building_tax_efficient
		trigger = { ROOT = { OR = { temple = 1 cathedral = 1 } } }
		tooltip = { custom_tooltip = building_tax_efficient_TT }
		delete_building_and_call_event_effect_efficient = yes
	}

	# Coastal defence chain
	option = {
		name = building_coastal_defence_all
		trigger = { ROOT = { OR = { coastal_defence = 1 naval_battery = 1 } } }
		tooltip = { custom_tooltip = building_coastal_defence_all_TT }
		delete_building_and_call_event_effect = { BUILDING = coastal_defence BUILDING = naval_battery }
	}
	
	# Shipyard chain
	option = {
		name = building_shipyard_all
		trigger = { ROOT = {  OR = { shipyard = 1 grand_shipyard = 1 } } }
		tooltip = { custom_tooltip = building_shipyard_all_TT }
		delete_building_and_call_event_effect = { BUILDING = shipyard BUILDING = grand_shipyard }
	}

	# Dock chain
	option = {
		name = building_dock_all
		trigger = { ROOT = { OR = { dock = 1 drydock = 1 } } }
		tooltip = { custom_tooltip = building_dock_all_TT }
		delete_building_and_call_event_effect = { BUILDING = dock BUILDING = drydock }
	}

	option = {
		name = building.1.exit
	}
}
