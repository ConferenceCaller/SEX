#A Slavic Alliance
country_event = {
	id = ME_Bulgaria_Events.2
	title = ME_Bulgaria_Events.2.T
	desc = ME_Bulgaria_Events.2.D
	picture = ORTHODOX_eventPicture

	is_triggered_only = yes

	option = {
		name = ME_Bulgaria_Events.2.OPT1

		ai_chance = {
			factor = 100
		}

		custom_tooltip = BUL_SLAVIC_ALLIANCE_TT
		every_country = {
			limit = {
				OR = {
					culture_group = east_slavic
					culture_group = west_slavic
					culture_group = south_slavic
				}
				total_development = 250
				OR = {
					alliance_with = ROOT
					has_opinion = {
						who = ROOT
						value = 100
					}
				}
			}
			add_historical_friend = ROOt
			ROOT = {
				add_historical_friend = PREV
			}
		}
	}
}

#Future of the Tsardom
country_event = {
	id = ME_Bulgaria_Events.3
	title = ME_Bulgaria_Events.3.T
	desc = ME_Bulgaria_Events.3.D
	picture = REVOLT_eventPicture

	is_triggered_only = yes

	option = {
		name = ME_Bulgaria_Events.3.OPT1

		ai_chance = {
			factor = 50
		}
		
		add_prestige_or_monarch_power = { amount = 25 }
		add_country_modifier = {
			name = bul_outward_look
			duration = 3650
		}
		change_primary_culture = greek
		custom_tooltip = BM_bul_legacy_of_simeon_2_TT
		hidden_effect = {
			every_owned_province = {
				limit = {
					culture = bulgarian
				}
				change_culture = greek
			}
		}
		change_tag = BYZ
		swap_non_generic_missions = yes
		on_change_tag_effect = yes
		add_country_modifier = {
			name = "centralization_modifier"
			duration = 7300
		}
		BYZ = {
			add_core = 151 # Constantinople
		}
		if = {
			limit = {
				is_part_of_hre = yes
			}
			every_owned_province = {
				limit = {
					is_part_of_hre = yes
				}
				set_in_empire = no
			}
		}
		if = {
			limit = { has_global_flag = hagia_sophia_now_mosque }
			clr_global_flag = hagia_sophia_now_mosque
			show_ambient_object = hagia_sophia
			hide_ambient_object = hagia_sophia_minarets
		}
		if = {
			limit = { NOT = { government = monarchy } }
			change_government = monarchy
		}
		if = {
			limit = { has_not_locked_first_tier_reform = yes }
			add_government_reform = byzantine_autocracy_reform
		}
		set_government_rank = 3
		if = {
			limit = { has_custom_ideas = no }
			country_event = { id = ideagroups.1 } #Swap Ideas
		}
		set_country_flag = formed_byzantium_flag
		change_ruler_stat = {
			type = adm
			amount = 1
		}
		change_ruler_stat = {
			type = dip
			amount = 1
		}
		change_ruler_stat = {
			type = mil
			amount = 1
		}
		if = {
			limit = {
				has_country_flag = BYZ_accepted_ottoman_threat
			}
			clr_country_flag = BYZ_accepted_ottoman_threat 
		}
		add_adm_power = 100
		add_dip_power = 100
		add_mil_power = 100
	}

	option = {
		name = ME_Bulgaria_Events.3.OPT2

		ai_chance = {
			factor = 50
		}
		
		add_legitimacy_or_mil_power = { amount = 25 }
		add_country_modifier = {
			name = bul_inward_look
			duration = -1
		}
		custom_tooltip = BM_bul_legacy_of_simeon_1_TT
		tooltip = {
			add_government_reform = nationalized_cetas_reform_upgraded
			add_government_reform = krums_code_reform_upgraded
		}
		hidden_effect = {
			set_country_flag = BM_BULGARIA_upgraded_gov_reforms_flag
			if = {
				limit = { 
					has_reform = nationalized_cetas_reform
				}
				add_government_reform = nationalized_cetas_reform_upgraded
			}
			if = {
				limit = { 
					has_reform = krums_code_reform
				}
				add_government_reform = krums_code_reform_upgraded
			}
		}
	}
}
