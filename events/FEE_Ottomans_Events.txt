#The Ottomans seize wheat from MOL or WAL

country_event = {
	id = FEE_Ottomans_Events.1
	title = FEE_Ottomans_Events.1.T
	desc = FEE_Ottomans_Events.1.D
	picture = FAMINE_eventPicture

	mean_time_to_happen = {
		months = 90
	}

	trigger = {
		is_at_war = no
		ai = yes
		tag = TUR

			AND = {
				stability = 3

					estate = all

				}

			}
		}
	}

	immediate = {
		if = {
			limit = { 
				adm_tech = 19
			}
			random_owned_province = {
				limit = {
					NOT = {
						has_tax_building_trigger = yes
					}
					can_build = cathedral
				}
				add_building = cathedral
			}
		}
		else_if = {
			limit = { 
				adm_tech = 4
			}
			random_owned_province = {
				limit = {
					NOT = {
						has_tax_building_trigger = yes
					}
					can_build = temple
				}
				add_building = temple
			}
		}
		if = {
			limit = { 
				dip_tech = 22
			}
			random_owned_province = {
				limit = {
					NOT = {
						has_trade_building_trigger = yes
					}
					can_build = stock_exchange
					OR = {
						has_river_estuary_trigger = yes
						province_has_center_of_trade_of_level = 1
					}
				}
				add_building = stock_exchange
			}
		}
		else_if = {
			limit = { 
				dip_tech = 17
			}
			random_owned_province = {
				limit = {
					NOT = {
						has_trade_building_trigger = yes
					}
					can_build = trade_depot
					OR = {
						has_river_estuary_trigger = yes
						province_has_center_of_trade_of_level = 1
					}
				}
				add_building = trade_depot
			}
		}
		else_if = {
			limit = { 
				dip_tech = 4
			}
			random_owned_province = {
				limit = {
					NOT = {
						has_trade_building_trigger = yes
					}
					can_build = marketplace
					OR = {
						has_river_estuary_trigger = yes
						province_has_center_of_trade_of_level = 1
					}
				}
				add_building = marketplace
			}
		}
		if = {
			limit = { 
				mil_tech = 16
			}
			random_owned_province = {
				limit = {
					NOT = {
						has_manpower_building_trigger = yes
					}
					can_build = training_fields
				}
				add_building = training_fields
			}
		}
		else_if = {
			limit = { 
				mil_tech = 6
			}
			random_owned_province = {
				limit = {
					NOT = {
						has_manpower_building_trigger = yes
					}
					can_build = barracks
				}
				add_building = barracks
			}
		}
		if = {
			limit = { 
				adm_tech = 22
			}
			random_owned_province = {
				limit = {
					NOT = {
						has_courthouse_building_trigger = yes
					}
				}
				add_building = town_hall
			}
		}
		else_if = {
			limit = { 
				adm_tech = 8
			}
			random_owned_province = {
				limit = {
					NOT = {
						has_courthouse_building_trigger = yes
					}
				}
				add_building = courthouse
			}
		}
	}

	option = {
		name = FEE_Ottomans_Events.1.OPT1
		trigger = {
			NOT = {
				stability = 3
			}
		}
		add_stability_or_adm_power = yes
		add_yearly_manpower = 0.1
		add_estate_loyalty = {
			estate = all
			loyalty = 5
		}
		add_years_of_income = 0.1
	}

	option = {
		name = FEE_Ottomans_Events.1.OPT1
		trigger = {
			NOT = {
				manpower_percentage = 1
			}
		}
		add_stability_or_adm_power = yes
		add_yearly_manpower = 0.5
		add_estate_loyalty = {
			estate = all
			loyalty = 5
		}
		add_years_of_income = 0.1
	}

	option = {
		name = FEE_Ottomans_Events.1.OPT1
		trigger = {
			NOT = {
				estate_loyalty = {
					estate = all
					loyalty = 30
				}
			}

		name = FEE_Ottomans_Events.2.OPT2

		134 = {
			add_base_production = 1
		}
	}
}

#Murad II's Return to the Throne
country_event = {
	id = FEE_Ottomans_Events.3
	title = FEE_Ottomans_Events.3.T
	desc = FEE_Ottomans_Events.3.D
	picture = COURT_eventPicture

	fire_only_once = yes

	trigger = {
		tag = TUR
		has_ruler = "Mehmet II Fatih"
		NOT = { is_year = 1450 }
		NOT = { stability = 2 }
		is_subject = no
	}

	mean_time_to_happen = {
		months = 24
	}

	immediate = {
		hidden_effect = {
			set_country_flag = fee_tur_murad_reign
		}
	}

	option = {
		name = FEE_Ottomans_Events.3.OPT1
		ai_chance = {
			factor = 100
		}

		define_ruler = {
			name = "Murad II"
			dynasty = Osmanoglu
			adm = 4
			dip = 3
			mil = 4
			age = 42
		}
		add_ruler_personality = calm_personality
		add_ruler_personality = well_connected_personality
		define_heir = {
			name = "Mehmet II Fatih"
			dynasty = Osmanoglu
			birth_date = 1432.3.30
			adm = 6
			dip = 5
			mil = 6
		}
		add_adm_power = 50
		add_country_modifier = {
			name = fee_tur_jannisariescalmed
			duration = 3650
		}
	}

	option = {
		name = FEE_Ottomans_Events.3.OPT2
		ai_chance = {
			factor = 0
		}

		add_stability = -1
		add_country_modifier = {
			name = fee_tur_jannisariesagainst_ruler
			duration = 3650
		}
		capital_scope = {
			noble_rebels = 2
		}
	}
}
