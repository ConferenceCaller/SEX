	[[amount]
		tooltip = {
			add_navy_tradition = $amount$
		}
		custom_tooltip = ME_ADD_NAVY_TRADITION_OR_DIP_POWER
		hidden_effect = {
			for = {
				amount = $amount$
				effect = "
					if = {
						limit = {
							NOT = { navy_tradition = 99.9 }
						}
						add_navy_tradition = 1
					}
					else = {
						add_dip_power = 3
					}
				"
			}
		}
	]

on_completed_agenda_effect = {
	on_completed_agenda_effect_$estate$ = yes
	if = {
		limit = {
			has_country_flag = BM_buffed_agendas_loyalty
		}
		add_estate_loyalty = {
			estate = all
			loyalty = 5
		}
	}
	if = {
		limit = { 
			has_government_attribute = enable_diverse_estates_reform
		}
		random_owned_province = {
			limit = {
				OR = {
					culture = czech
					culture = czech_new
					culture = hungarian
					culture_group = germanic
				}
			}
			add_random_development = 1
		}
	}
	clr_auto_complete_flag = yes
}
